cmake_minimum_required(VERSION 3.20)
project(VibeBackup VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(VibeBackup WIN32 src/main.cpp)
target_compile_definitions(VibeBackup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(VibeBackup PRIVATE comctl32 shell32 ole32 advapi32)
if(MSVC)
  target_compile_options(VibeBackup PRIVATE /W4 /EHsc /utf-8)
endif()
